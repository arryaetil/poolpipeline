- [ ] **Step 1: Add failing test**

Add at the bottom of `tests/server.test.js`:

```js
test('GET /demo-runs route is registered', () => {
  assert.equal(hasRoute('get', '/demo-runs'), true);
});

test('GET /demo-runs returns an array', async () => {
  const server = app.listen(0, '127.0.0.1');
  await new Promise((resolve) => server.once('listening', resolve));
  try {
    const { port } = server.address();
    const res = await fetch(`http://127.0.0.1:${port}/demo-runs`);
    const body = await res.json();
    assert.equal(res.status, 200);
    assert.ok(Array.isArray(body));
  } finally {
    await new Promise((resolve, reject) => server.close((err) => err ? reject(err) : resolve()));
  }
});
```

- [ ] **Step 3: Add readdirSync + existsSync to imports in server.js**

```js

```

to:

```js

```

- [ ] **Step 4: Add /demo-runs route to server.js**

Add after the `app.get('/features', ...)` route (around line 38):

```js
app.get('/demo-runs', (_req, res) => {
  const outputDir = join(__dirname, 'output');
  if (!existsSync(outputDir)) return res.json([]);
  const runs = readdirSync(outputDir, { withFileTypes: true })
    .filter((d) => d.isDirectory())
    .map((d) => d.name)
    .filter((name) => existsSync(join(outputDir, name, 'edited.png')));
  res.json(runs);
});
```

- [ ] **Step 5: Run tests to verify they pass**

```bash
node --test tests/server.test.js
```

Expected: all tests pass. The output/ directory has several completed runs so the array will be non-empty.

```bash

git commit -m "feat: add GET /demo-runs endpoint"
```

// Progress steps
const STEP_IDS = ['geocode', 'satellite', 'analysis', 'renders', 'view_select', 'edit'];

function setStep(id, state) {
  const dot = document.getElementById(`dot-${id}`);
  const txt = document.getElementById(`txt-${id}`);
  if (!dot || !txt) return;
  dot.className = `step-dot dot-${state}`;
  dot.innerHTML = state === 'done' ? '<i data-lucide="check"></i>' : '';
  txt.className = `step-text step-text-${state}`;
  if (state === 'done') lucide.createIcons({ nodes: [dot] });
}

function resetSteps() {
  rendersCount = 0;
  Object.keys(renderStore).forEach((k) => delete renderStore[k]);
  STEP_IDS.forEach((id) => setStep(id, 'pending'));
  document.getElementById('txt-renders').textContent = '3D renders gemaakt';
  document.getElementById('progressError').innerHTML = '';
}

// SSE handlers
function handleStep({ step, status, value }) {
  if (step === 'geocode') { setStep('geocode', 'done'); setStep('satellite', 'active'); }
  if (step === 'satellite') { setStep('satellite', 'done'); setStep('analysis', 'active'); }
  if (step === 'analysis') { setStep('analysis', 'done'); setStep('renders', 'active'); }
  if (step === 'renders') {
    if (value?.image) renderStore[value.heading] = value.image;
    rendersCount += 1;
    document.getElementById('txt-renders').textContent = `3D renders gemaakt (${rendersCount} / 4)`;
    if (rendersCount === 4) { setStep('renders', 'done'); setStep('view_select', 'active'); }
  }
  if (step === 'view_select') {
    setStep('view_select', 'done');
    setStep('edit', 'active');
    chosenRenderSrc = renderStore[value.chosen_heading] ?? null;
  }
  if (step === 'edit') {
    setStep('edit', 'done');
    editedSrc = value;
    showResult(chosenRenderSrc, editedSrc);
  }
}

function handleError({ step, message }) {
  document.getElementById('progressError').innerHTML = `<div class="error-msg">${escHtml(message)}</div>`;
  if (step) setStep(step, 'pending');
  document.getElementById('genBtn').disabled = false;
}

// Run pipeline
async function run() {
  if (demoMode) return;
  const address = document.getElementById('addrInput').value.trim();
  if (!address) { alert('Voer een adres in'); return; }
  if (!selectedFeatureId) { alert('Kies een product'); return; }

  resetSteps();
  chosenRenderSrc = null;
  editedSrc = null;
  document.getElementById('progressCard').classList.remove('hidden');
  document.getElementById('resultCard').classList.add('hidden');
  document.getElementById('genBtn').disabled = true;
  setStep('geocode', 'active');

  try {
    const res = await fetch('/run', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ address, feature: selectedFeatureId }),
    });
    if (!res.ok) { const err = await res.json(); throw new Error(err.error || `Fout ${res.status}`); }

    const reader = res.body.getReader();
    const decoder = new TextDecoder();
    let buffer = '';
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });
      const events = buffer.split('\n\n');
      buffer = events.pop();
      for (const raw of events) {
        let eventName = null, dataLine = null;
        for (const line of raw.split('\n')) {
          if (line.startsWith('event: ')) eventName = line.slice(7).trim();
          if (line.startsWith('data: ')) dataLine = line.slice(6);
        }
        if (!eventName || dataLine === null) continue;
        const data = JSON.parse(dataLine);
        if (eventName === 'step') handleStep(data);
        if (eventName === 'error') handleError(data);
      }
    }
  } catch (err) {
    handleError({ step: 'geocode', message: err.message });
  } finally {
    document.getElementById('genBtn').disabled = false;
  }
}

// Before/after slider
function initSlider() {
  const container = document.getElementById('sliderContainer');
  const handle = document.getElementById('baHandle');
  const afterImg = document.getElementById('imgAfter');
  const divider = document.getElementById('baDivider');
  let dragging = false;

  function setPos(clientX) {
    const rect = container.getBoundingClientRect();
    const pct = Math.max(0, Math.min(100, (clientX - rect.left) / rect.width * 100));
    afterImg.style.clipPath = `inset(0 0 0 ${pct}%)`;
    handle.style.left = `${pct}%`;
    divider.style.left = `${pct}%`;
  }

  handle.addEventListener('mousedown', (e) => { dragging = true; e.preventDefault(); });
  document.addEventListener('mouseup', () => { dragging = false; });
  document.addEventListener('mousemove', (e) => { if (dragging) setPos(e.clientX); });
  handle.addEventListener('touchstart', (e) => { dragging = true; e.preventDefault(); }, { passive: false });
  document.addEventListener('touchend', () => { dragging = false; });
  document.addEventListener('touchmove', (e) => { if (dragging) setPos(e.touches[0].clientX); }, { passive: true });
  container.addEventListener('click', (e) => { if (e.target !== handle) setPos(e.clientX); });
}
